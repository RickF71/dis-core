// src/ws/command/handler.rs

use std::sync::{Arc, Mutex};

use warp::ws::{Message, WebSocket};
use futures_util::{StreamExt, SinkExt};

use crate::runtime::node_runtime::NodeRuntime;
use crate::ws::command::messages::{
    CommandMessage,
    CommandResponse,
};

pub async fn handle_command_ws(
    socket: WebSocket,
    _node_runtime: Arc<Mutex<NodeRuntime>>,
) {
    let (mut tx, mut rx) = socket.split();

    while let Some(Ok(msg)) = rx.next().await {
        if !msg.is_text() {
            continue;
        }

        let parsed = serde_json::from_str::<CommandMessage>(
            msg.to_str().unwrap(),
        );

        let response = match parsed {
            Ok(_command) => {
                // NOTE: no authority, no mutation yet
                CommandResponse::Accepted {
                    intent_id: "intent-temp-id".to_string(),
                }
            }
            Err(err) => {
                CommandResponse::Denied {
                    intent_id: "unknown".to_string(),
                    reason: format!("invalid command: {}", err),
                }
            }
        };

        let _ = tx
            .send(Message::text(
                serde_json::to_string(&response).unwrap(),
            ))
            .await;
    }
}

use warp::ws::{Message, WebSocket};
use futures_util::{SinkExt, StreamExt};
use std::time::Duration;
use tokio::time::sleep;
use std::sync::{Arc, Mutex};

use crate::runtime::node_runtime::NodeRuntime;
use crate::ws::observe::messages::*;

pub async fn handle_observe_ws(
    socket: WebSocket,
    _node_runtime: Arc<Mutex<NodeRuntime>>,
) {
    let (mut tx, _) = socket.split();

    let mut tick: u64 = 0;

    loop {
        tick += 1;

        let snapshot = ObserveMessage::Snapshot(ObserveSnapshot {
            tick,
            node: NodeView {
                name: "dis-core",
                version: "0.1.0",
            },
            domain: None,
            spine: SpineView {
                nullus: true,
                aether: false,
                terra: false,
                numen: false,
                lima: false,
                corporeal: false,
            },
            capabilities: CapabilityView {
                can_bind_identity: false,
                can_claim_sovereignty: false,
            },
        });

        if tx
            .send(Message::text(
                serde_json::to_string(&snapshot).unwrap(),
            ))
            .await
            .is_err()
        {
            break;
        }

        sleep(Duration::from_millis(500)).await;
    }
}

use std::sync::{Arc, Mutex};
use warp::Filter;

use crate::runtime::node_runtime::NodeRuntime;
use crate::ws::observe::handler::handle_observe_ws;
use crate::ws::command::handler::handle_command_ws;
use crate::ws::totem::handler::handle_totem_ws;

pub fn ws_routes(
    node_runtime: Arc<Mutex<NodeRuntime>>,
) -> impl Filter<Extract = impl warp::Reply, Error = warp::Rejection> + Clone {

    // Inject NodeRuntime into all WS handlers
    let runtime = warp::any().map(move || node_runtime.clone());

    // ---------------------------------------------
    // Observer WebSocket (read-only, structure)
    // ws://host/ws/observe
    // ---------------------------------------------
    let observe = warp::path!("ws" / "observe")
        .and(warp::ws())
        .and(runtime.clone())
        .map(|ws: warp::ws::Ws, runtime| {
            ws.on_upgrade(move |socket| handle_observe_ws(socket, runtime))
        });

    // ---------------------------------------------
    // Command WebSocket (intent, input)
    // ws://host/ws/command
    // ---------------------------------------------
    let command = warp::path!("ws" / "command")
        .and(warp::ws())
        .and(runtime.clone())
        .map(|ws: warp::ws::Ws, runtime| {
            ws.on_upgrade(move |socket| handle_command_ws(socket, runtime))
        });

    // ---------------------------------------------
    // Totem WebSocket (presence)
    // ws://host/ws/totem
    // ---------------------------------------------
    let totem = warp::path!("ws" / "totem")
        .and(warp::ws())
        .and(runtime)
        .map(|ws: warp::ws::Ws, runtime| {
            ws.on_upgrade(move |socket| handle_totem_ws(socket, runtime))
        });

    observe.or(command).or(totem)
}

use std::sync::{Arc, Mutex};

use futures_util::StreamExt;
use warp::ws::WebSocket;

use crate::runtime::node_runtime::NodeRuntime;
use crate::ws::totem::messages::TotemInbound;

/// Handle the Totem WebSocket connection.
///
/// Responsibilities:
/// - Receive heartbeat messages from totemd
/// - Update NodeRuntime.totem presence
/// - Clear presence on disconnect
///
/// Non-responsibilities:
/// - No broadcasting
/// - No authority decisions
/// - No timestamps or formatting
/// - No watchdog timers
pub async fn handle_totem_ws(
    mut socket: WebSocket,
    node_runtime: Arc<Mutex<NodeRuntime>>,
) {
    // Process incoming WS messages
    while let Some(Ok(msg)) = socket.next().await {
        if !msg.is_text() {
            continue;
        }

        let text = match msg.to_str() {
            Ok(t) => t,
            Err(_) => continue,
        };

        // Parse inbound totem messages
        match serde_json::from_str::<TotemInbound>(text) {
            Ok(TotemInbound::Heartbeat(_)) => {
                let mut runtime = node_runtime
                    .lock()
                    .expect("NodeRuntime mutex poisoned");

                runtime.totem_mut().heartbeat();
            }

            // Unknown / malformed messages are ignored
            Err(_) => {}
        }
    }

    // WebSocket closed → totem is immediately offline
    {
        let mut runtime = node_runtime
            .lock()
            .expect("NodeRuntime mutex poisoned");

        runtime.totem_mut().clear();
    }
}
